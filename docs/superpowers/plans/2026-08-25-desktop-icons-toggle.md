# DesktopIconsToggle Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a resident Windows tray utility that hides/shows desktop icons on a middle-button double-click on the desktop, and via a notification-area icon that only exists while the desktop is in the foreground.

**Architecture:** One process, one UI thread owning a message-only window and its message loop. A global low-level mouse hook and a WinEvent foreground hook post messages to that window; the window procedure is the only place that acts. Win32 is reached through hand-written P/Invoke — no WinForms — so the app publishes as a NativeAOT single file.

**Tech Stack:** C# / .NET 10 (`net10.0-windows`), NativeAOT, xUnit for the two pieces of pure logic, PowerShell for icon generation.

**Spec:** `docs/superpowers/specs/2026-08-25-desktop-icons-toggle-design.md`

## Global Constraints

- Target framework: `net10.0-windows`, RID `win-x64`. Nothing else.
- `PublishAot=true` on the app project. No dependency that breaks AOT; no reflection over app types.
- No WinForms, no WPF, no `System.Windows.Forms` reference.
- `Nullable=enable` and `TreatWarningsAsErrors=true` on both projects.
- The desktop-icons toggle message is `WM_COMMAND` with `wParam = 0x7402` posted to `SHELLDLL_DefView`.
- Icon visibility is read from `IsWindowVisible` on the `SysListView32` child — **never** from the `HideIcons` registry value.
- The mouse-hook callback never calls into Explorer and never allocates on the fast path; it only posts `WM_APP_TOGGLE`.
- On every exit path the process restores the icons if they are hidden.
- All user-visible strings (menu, tooltip) are French.
- During Tasks 1-8 the app project is `<OutputType>Exe</OutputType>` (console) so verification steps can print. Task 9 flips it to `WinExe`.
- The installed SDK is 10.0.400; MSVC 14.51.36231 and Windows SDK 10.0.26100 are already present.

Create `tools/make-icon.ps1`. It draws a 64x64 desktop glyph (frame plus four tiles) and wraps the PNG in a one-entry ICO container:

```powershell
Add-Type -AssemblyName System.Drawing
$size = 64
$bmp  = New-Object System.Drawing.Bitmap($size, $size)
$g    = [System.Drawing.Graphics]::FromImage($bmp)
$g.SmoothingMode = 'AntiAlias'
$g.Clear([System.Drawing.Color]::Transparent)

$pen = New-Object System.Drawing.Pen([System.Drawing.Color]::FromArgb(255, 235, 235, 235), 4)
$g.DrawRectangle($pen, 6, 6, ($size - 13), ($size - 13))

$brush = New-Object System.Drawing.SolidBrush([System.Drawing.Color]::FromArgb(255, 120, 180, 255))
foreach ($p in @(@(18, 18), @(36, 18), @(18, 36), @(36, 36))) {
    $g.FillRectangle($brush, $p[0], $p[1], 10, 10)
}
$g.Dispose()

$ms = New-Object System.IO.MemoryStream
$bmp.Save($ms, [System.Drawing.Imaging.ImageFormat]::Png)
$png = $ms.ToArray()
$bmp.Dispose()

$out = New-Object System.IO.MemoryStream
$w   = New-Object System.IO.BinaryWriter($out)
$w.Write([UInt16]0); $w.Write([UInt16]1); $w.Write([UInt16]1)          # ICONDIR
$w.Write([Byte]$size); $w.Write([Byte]$size); $w.Write([Byte]0); $w.Write([Byte]0)
$w.Write([UInt16]1); $w.Write([UInt16]32)
$w.Write([UInt32]$png.Length); $w.Write([UInt32]22)                    # ICONDIRENTRY
$w.Write($png)
$w.Flush()

$dir = Join-Path $PSScriptRoot '..\src\DesktopIconsToggle'
[System.IO.File]::WriteAllBytes((Join-Path (Resolve-Path -LiteralPath $dir).Path 'app.ico'), $out.ToArray())
Write-Output "app.ico written"
```

```csharp
namespace DesktopIconsToggle;

/// <summary>
/// Recognises a middle-button double click. Deliberately free of Win32 so it
/// can be tested against an injected clock.
/// </summary>
internal sealed class ClickDetector
{
    private readonly long _windowMs;
    private long _lastClickMs;
    private bool _armed;

    public ClickDetector(int doubleClickTimeMs)
        => _windowMs = (long)(doubleClickTimeMs * 1.5);

    /// <summary>Returns true on the click that completes a pair.</summary>
    public bool RegisterClick(long nowMs)
    {
        if (_armed && nowMs - _lastClickMs <= _windowMs)
        {
            _armed = false;
            return true;
        }

        _armed = true;
        _lastClickMs = nowMs;
        return false;
    }
}
```

```csharp
/// <summary>The only component that talks to Explorer.</summary>
internal static class IconToggler
{
    private static readonly Win32.EnumWindowsProc EnumCallback = EnumWorkerW;
    private static IntPtr _workerDefView;

    public static IntPtr FindDefView()
    {
        IntPtr progman = Win32.FindWindowW("Progman", null);
        if (progman != IntPtr.Zero)
        {
            IntPtr direct = Win32.FindWindowExW(progman, IntPtr.Zero, "SHELLDLL_DefView", null);
            if (direct != IntPtr.Zero)
            {
                return direct;
            }
        }

        // Animated wallpapers and some multi-monitor setups reparent the view
        // under a top-level WorkerW instead.
        _workerDefView = IntPtr.Zero;
        Win32.EnumWindows(EnumCallback, IntPtr.Zero);
        return _workerDefView;
    }

    private static bool EnumWorkerW(IntPtr hWnd, IntPtr lParam)
    {
        if (Win32.GetClassName(hWnd) != "WorkerW")
        {
            return true;
        }

        IntPtr child = Win32.FindWindowExW(hWnd, IntPtr.Zero, "SHELLDLL_DefView", null);
        if (child == IntPtr.Zero)
        {
            return true;
        }

        _workerDefView = child;
        return false;
    }

    /// <summary>
    /// Reads the live window state. The HideIcons registry value is not a
    /// reliable mirror of the toggle message, so it is never consulted.
    /// </summary>
    public static bool AreIconsVisible()
    {
        IntPtr defView = FindDefView();
        if (defView == IntPtr.Zero)
        {
            return true;
        }

        IntPtr list = Win32.FindWindowExW(defView, IntPtr.Zero, "SysListView32", null);
        return list != IntPtr.Zero && Win32.IsWindowVisible(list);
    }

    public static bool Toggle()
    {
        IntPtr defView = FindDefView();
        if (defView == IntPtr.Zero)
        {
            return false;
        }

        return Win32.PostMessageW(defView, Win32.WM_COMMAND,
                                  new IntPtr(Win32.TOGGLE_DESKTOP_ICONS), IntPtr.Zero);
    }

    /// <summary>Safety net used on every exit path.</summary>
    public static void EnsureVisible()
    {
        if (!AreIconsVisible())
        {
            Toggle();
        }
    }
```

```csharp
using DesktopIconsToggle.Native;

namespace DesktopIconsToggle;

/// <summary>
/// Raises <see cref="Changed"/> when the desktop gains or loses the foreground.
/// Event-driven: no timer, no polling, no wake-ups while nothing happens.
/// </summary>
internal sealed class DesktopWatcher : IDisposable
{
    // Held in a field so the GC never collects the callback the OS holds.
    private readonly Win32.WinEventProc _callback;
    private IntPtr _hook;

    public DesktopWatcher() => _callback = OnForegroundChanged;

    public event Action<bool>? Changed;

    public bool IsDesktopForeground { get; private set; }

    public static bool IsDesktopWindow(IntPtr hWnd)
    {
        if (hWnd == IntPtr.Zero)
        {
            return false;
        }

        string className = Win32.GetClassName(hWnd);
        return className is "Progman" or "WorkerW";
    }

    public void Start()
    {
        IsDesktopForeground = IsDesktopWindow(Win32.GetForegroundWindow());
        _hook = Win32.SetWinEventHook(Win32.EVENT_SYSTEM_FOREGROUND, Win32.EVENT_SYSTEM_FOREGROUND,
                                      IntPtr.Zero, _callback, 0, 0, Win32.WINEVENT_OUTOFCONTEXT);
        Changed?.Invoke(IsDesktopForeground);
    }

    private void OnForegroundChanged(IntPtr hWinEventHook, uint eventType, IntPtr hwnd,
                                     int idObject, int idChild, uint dwEventThread, uint dwmsEventTime)
    {
        // OBJID_WINDOW == 0; ignore per-control focus noise.
        if (idObject != 0)
        {
            return;
        }

        bool isDesktop = IsDesktopWindow(hwnd);
        if (isDesktop == IsDesktopForeground)
        {
            return;
        }

        IsDesktopForeground = isDesktop;
        Changed?.Invoke(isDesktop);
    }

    public void Dispose()
    {
        if (_hook != IntPtr.Zero)
        {
            Win32.UnhookWinEvent(_hook);
            _hook = IntPtr.Zero;
        }
    }
}
```

```csharp
using System.Runtime.InteropServices;
using DesktopIconsToggle.Native;

namespace DesktopIconsToggle;

/// <summary>
/// Global low-level mouse hook watching for a middle double-click over the desktop.
///
/// The hook stays installed for the whole process lifetime. Installing it only
/// while the desktop has the foreground would drop the first click, because that
/// click is what gives the desktop the foreground in the first place. The cost of
/// staying installed is one integer comparison per mouse event.
/// </summary>
internal sealed class MouseHook : IDisposable
{
    private static readonly string[] DesktopClasses =
        ["Progman", "WorkerW", "SHELLDLL_DefView", "SysListView32"];

    private readonly Win32.HookProc _callback;   // kept alive for the OS
    private readonly ClickDetector _detector;
    private readonly IntPtr _targetWindow;
    private IntPtr _hook;

    public MouseHook(IntPtr targetWindow)
    {
        _targetWindow = targetWindow;
        _callback = OnMouseEvent;
        _detector = new ClickDetector((int)Win32.GetDoubleClickTime());
    }

    /// <summary>Diagnostics only: fires on every completed middle double-click.</summary>
    public Action<bool>? Trace { get; set; }

    public bool Install()
    {
        _hook = Win32.SetWindowsHookExW(Win32.WH_MOUSE_LL, _callback,
                                        Win32.GetModuleHandleW(null), 0);
        return _hook != IntPtr.Zero;
    }

    private IntPtr OnMouseEvent(int nCode, IntPtr wParam, IntPtr lParam)
    {
        // Fast path: everything that is not a middle press leaves immediately,
        // with no allocation and no marshalling.
        if (nCode < 0 || (int)wParam != Win32.WM_MBUTTONDOWN)
        {
            return Win32.CallNextHookEx(_hook, nCode, wParam, lParam);
        }

        if (_detector.RegisterClick(Win32.GetTickCount()))
        {
            var data = Marshal.PtrToStructure<Win32.MSLLHOOKSTRUCT>(lParam);
            bool overDesktop = IsOverDesktop(data.pt);
            Trace?.Invoke(overDesktop);

            if (overDesktop)
            {
                // Never call into Explorer from a hook: Windows unhooks slow callbacks.
                Win32.PostMessageW(_targetWindow, Win32.WM_APP_TOGGLE, IntPtr.Zero, IntPtr.Zero);
            }
        }

        // Always let the click through; middle-clicking the desktop does nothing natively.
        return Win32.CallNextHookEx(_hook, nCode, wParam, lParam);
    }

    private static bool IsOverDesktop(Win32.POINT point)
    {
        IntPtr hWnd = Win32.WindowFromPoint(point);
        for (int depth = 0; hWnd != IntPtr.Zero && depth < 8; depth++)
        {
            if (Array.IndexOf(DesktopClasses, Win32.GetClassName(hWnd)) >= 0)
            {
                return true;
            }

            hWnd = Win32.GetAncestor(hWnd, Win32.GA_PARENT);
        }

        return false;
    }

    public void Dispose()
    {
        if (_hook != IntPtr.Zero)
        {
            Win32.UnhookWindowsHookEx(_hook);
            _hook = IntPtr.Zero;
        }
    }
}
```

```csharp
using DesktopIconsToggle.Native;

namespace DesktopIconsToggle;

/// <summary>Hidden HWND_MESSAGE window: the single place where work happens.</summary>
internal sealed class AppWindow : IDisposable
{
    private const string ClassName = "DesktopIconsToggleWnd";

    private readonly Win32.WndProc _wndProc;    // kept alive for the OS
    private readonly IntPtr _classNamePtr;

    public AppWindow()
    {
        _wndProc = WindowProc;
        _classNamePtr = Marshal.StringToHGlobalUni(ClassName);

        var wc = new Win32.WNDCLASSEXW
        {
            cbSize = (uint)Marshal.SizeOf<Win32.WNDCLASSEXW>(),
            lpfnWndProc = Marshal.GetFunctionPointerForDelegate(_wndProc),
            hInstance = Win32.GetModuleHandleW(null),
            lpszClassName = _classNamePtr,
        };

        if (Win32.RegisterClassExW(ref wc) == 0)
        {
            throw new InvalidOperationException(
                $"RegisterClassEx failed: {Marshal.GetLastWin32Error()}");
        }

        Handle = Win32.CreateWindowExW(0, ClassName, ClassName, 0, 0, 0, 0, 0,
                                       Win32.HWND_MESSAGE, IntPtr.Zero,
                                       Win32.GetModuleHandleW(null), IntPtr.Zero);

        if (Handle == IntPtr.Zero)
        {
            throw new InvalidOperationException(
                $"CreateWindowEx failed: {Marshal.GetLastWin32Error()}");
        }
    }

    public IntPtr Handle { get; }

    /// <summary>Return true to swallow the message.</summary>
    public event Func<uint, IntPtr, IntPtr, bool>? MessageReceived;

    private IntPtr WindowProc(IntPtr hWnd, uint msg, IntPtr wParam, IntPtr lParam)
    {
        if (MessageReceived?.Invoke(msg, wParam, lParam) == true)
        {
            return IntPtr.Zero;
        }

        if (msg == Win32.WM_DESTROY)
        {
            Win32.PostQuitMessage(0);
            return IntPtr.Zero;
        }

        return Win32.DefWindowProcW(hWnd, msg, wParam, lParam);
    }

    public void RunMessageLoop()
    {
        while (Win32.GetMessageW(out var msg, IntPtr.Zero, 0, 0) > 0)
        {
            Win32.TranslateMessage(ref msg);
            Win32.DispatchMessageW(ref msg);
        }
    }

    public void Dispose()
    {
        if (Handle != IntPtr.Zero)
        {
            Win32.DestroyWindow(Handle);
        }

        if (_classNamePtr != IntPtr.Zero)
        {
            Marshal.FreeHGlobal(_classNamePtr);
        }
    }
}
```

```csharp
        {
            return 0;
        }

        string exePath = Environment.ProcessPath ?? Process.GetCurrentProcess().MainModule!.FileName;
        _autostart = new Autostart(new RunKeyStore(), exePath);

        _taskbarCreatedMessage = Win32.RegisterWindowMessageW("TaskbarCreated");

        var window = new AppWindow();
        _window = window;
        window.MessageReceived += OnMessage;

        // Locals, not the nullable fields: a lambda capturing a nullable field
        // loses its flow analysis and trips CS8602 under TreatWarningsAsErrors.
        var tray = new TrayIcon(window.Handle);
        _tray = tray;
        tray.CommandChosen += OnCommand;

        var watcher = new DesktopWatcher();
        _watcher = watcher;
        watcher.Changed += isDesktop => tray.SetVisible(isDesktop);
        watcher.Start();

        var mouseHook = new MouseHook(window.Handle);
        _mouseHook = mouseHook;
        if (!mouseHook.Install())
        {
            tray.UpdateTooltip("Icônes du Bureau — raccourci souris indisponible");
        }
        else
        {
            RefreshTooltip();
        }

        AppDomain.CurrentDomain.ProcessExit += (_, _) => Shutdown();

        window.RunMessageLoop();

        Shutdown();
        return 0;
    }

    private static bool OnMessage(uint msg, IntPtr wParam, IntPtr lParam)
    {
        if (msg == Win32.WM_APP_TOGGLE)
        {
            ToggleIcons();
            return true;
        }

        if (msg == Win32.WM_APP_TRAY)
        {
            int mouseMessage = (int)lParam;
            if (mouseMessage == Win32.WM_LBUTTONUP)
            {
                ToggleIcons();
            }
            else if (mouseMessage == Win32.WM_RBUTTONUP)
            {
                _tray?.ShowMenu(IconToggler.AreIconsVisible(), _autostart?.IsEnabled() ?? false);
            }

            return true;
        }

        if (_taskbarCreatedMessage != 0 && msg == _taskbarCreatedMessage)
        {
            _tray?.Readd();
            return true;
        }

        if (msg == Win32.WM_ENDSESSION)
        {
            Shutdown();
            return false;
        }

        return false;
    }

    private static void OnCommand(int command)
    {
        switch (command)
        {
            case TrayIcon.CmdToggle:
                ToggleIcons();
                break;

            case TrayIcon.CmdAutostart:
                _autostart?.Toggle();
                break;

            case TrayIcon.CmdQuit:
                Win32.PostQuitMessage(0);
                break;
        }
    }

    private static void ToggleIcons()
    {
        if (!IconToggler.Toggle())
        {
            _tray?.UpdateTooltip("Icônes du Bureau — bureau indisponible");
            return;
        }

        // Explorer needs a beat before the window state reflects the change.
        Thread.Sleep(120);
        RefreshTooltip();
    }

    private static void RefreshTooltip()
        => _tray?.UpdateTooltip(IconToggler.AreIconsVisible()
            ? "Icônes du Bureau — affichées"
            : "Icônes du Bureau — masquées");

    private static void Shutdown()
    {
        IconToggler.EnsureVisible();   // never leave the user without icons
        _tray?.Dispose();
        _mouseHook?.Dispose();
        _watcher?.Dispose();
        _window?.Dispose();
        _tray = null;
        _mouseHook = null;
        _watcher = null;
        _window = null;
    }
}
```
